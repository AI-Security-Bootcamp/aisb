# Host setup for Exercise 7.3: NVIDIA Container Toolkit vulnerability
#
# Run only on a disposable lab host. The exercise intentionally installs a
# vulnerable GPU container stack.
#
# Known-good tested stack:
# - Ubuntu 22.04, Linux 5.15
# - NVIDIA A10, driver 580.173.02
# - Docker 29.1.3-0ubuntu3~22.04.2
# - containerd 2.2.1-0ubuntu1~22.04.2
# - runc 1.1.0-0ubuntu1
# - nvidia-container-toolkit 1.17.7-1
# - nvidia-container-toolkit-base 1.17.7-1
# - libnvidia-container-tools 1.17.7-1
# - libnvidia-container1 1.17.7-1
# - nvidia-container-runtime.mode = "legacy"
# - nvidia-container-runtime.modes.legacy.cuda-compat-mode = "hook"

set -eux

# Baseline checks: these should succeed before configuring the vulnerable stack.
nvidia-smi
docker --version

# Install the vulnerable NVIDIA Container Toolkit stack from the configured
# NVIDIA apt repository. If the repository is not configured, use the tarball
# fallback below.
sudo apt-mark unhold \
  nvidia-container-toolkit \
  nvidia-container-toolkit-base \
  libnvidia-container-tools \
  libnvidia-container1 || true

sudo DEBIAN_FRONTEND=noninteractive apt-get install -y --allow-downgrades \
  nvidia-container-toolkit=1.17.7-1 \
  nvidia-container-toolkit-base=1.17.7-1 \
  libnvidia-container-tools=1.17.7-1 \
  libnvidia-container1=1.17.7-1

sudo apt-mark hold \
  nvidia-container-toolkit \
  nvidia-container-toolkit-base \
  libnvidia-container-tools \
  libnvidia-container1

# Fallback if the apt repository does not expose 1.17.7:
#
# arch="$(dpkg --print-architecture)"
# wget "https://github.com/NVIDIA/nvidia-container-toolkit/releases/download/v1.17.7/nvidia-container-toolkit_1.17.7_deb_${arch}.tar.gz"
# tar xf "nvidia-container-toolkit_1.17.7_deb_${arch}.tar.gz"
# sudo dpkg -i "release-v1.17.7-stable/packages/ubuntu18.04/${arch}/"*.deb

# Pin runc to the version verified to preserve the hook inheritance behavior
# this lab demonstrates.
sudo apt-mark unhold runc || true
sudo DEBIAN_FRONTEND=noninteractive apt-get install -y --allow-downgrades \
  runc=1.1.0-0ubuntu1
sudo apt-mark hold runc

# Configure Docker to expose the NVIDIA runtime and force the vulnerable CUDA
# compatibility hook path.
sudo nvidia-ctk runtime configure --runtime=docker
sudo nvidia-ctk config --in-place \
  --set nvidia-container-runtime.mode=legacy \
  --set nvidia-container-runtime.modes.legacy.cuda-compat-mode=hook
sudo systemctl restart docker || {
  sudo systemctl reset-failed docker
  sudo systemctl start docker
}

# Verify host configuration.
nvidia-ctk --version
runc --version
docker info --format 'Default={{.DefaultRuntime}} Runtimes={{json .Runtimes}}'
grep -E 'mode =|cuda-compat-mode' /etc/nvidia-container-runtime/config.toml

# Test the exercise. The direct LD_PRELOAD test proves the payload works; the
# Docker test proves that the container image's LD_PRELOAD reached the
# privileged NVIDIA createContainer hook.
cd solution
make clean
make

rm -f /tmp/output
LD_PRELOAD=./poc.so nvidia-ctk --version
cat /tmp/output

rm -f /tmp/output
docker run --rm --runtime=nvidia --gpus=all nvidia-ctk-image echo hello
ls -l /tmp/output
cat /tmp/output
sudo rm -f /tmp/output
